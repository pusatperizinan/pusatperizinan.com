    83→/** Potong metaDesc agar tidak melebihi batas mesin pencari dengan rapi */
    84→function capMeta(text: string, max = 300): string {
    85→  if (text.length <= max) return text;
    86→  const cut = text.slice(0, max);
    87→  const lastStop = Math.max(cut.lastIndexOf(". "), cut.lastIndexOf("— "), cut.lastIndexOf(", "));
    88→  return (lastStop > max * 0.6 ? cut.slice(0, lastStop) : cut.replace(/[,;\s]+$/, "")) + ".";
    89→}

   657→// ------------------------------------------------------------
   658→// BUILD ALL — agregasi seluruh katalog
   659→// ------------------------------------------------------------
   660→
   661→function buildAllPages(): ServicePage[] {
   662→  const pages: ServicePage[] = [];
   663→
   664→  // 1. Halaman induk (61 layanan)
   665→  for (const id of BASE_SERVICES.keys()) {
   666→    const p = buildBasePage(id);
   667→    if (p) pages.push(p);
   668→  }
   669→
   670→  // 2. Perizinan × 38 provinsi (layanan populer)
   671→  for (const svc of REGION_FULL_SERVICES) {
   672→    for (let i = 0; i < PROVINCES.length; i++) {
   673→      const p = buildRegionPage(svc, i);
   674→      if (p) pages.push(p);
   675→    }
   676→  }
   677→  // 2b. Perizinan lain × 8 provinsi utama
   678→  const liteServices = [...BASE_SERVICES.values()]
   679→    .filter((b) => b.category === "perizinan" && !REGION_FULL_SERVICES.includes(b.id))
   680→    .map((b) => b.id);
   681→  for (const svc of liteServices) {
   682→    for (const provName of REGION_LITE_PROVINCES) {
   683→      const idx = PROVINCES.findIndex((p) => p.name === provName);
   684→      if (idx >= 0) {
   685→        const p = buildRegionPage(svc, idx);
   686→        if (p) pages.push(p);
   687→      }
   688→    }
   689→  }
   690→
   691→  // 2c. Pajak × 38 provinsi (layanan pajak paling dicari)
   692→  for (const svc of REGION_TAX_SERVICES) {
   693→    for (let i = 0; i < PROVINCES.length; i++) {
   694→      const p = buildRegionPage(svc, i);
   695→      if (p) pages.push(p);
   696→    }
   697→  }
   698→
   699→  // 2d. PMI B2C × 8 provinsi utama pengirim migran
   700→  for (const svc of REGION_PMI_SERVICES) {
   701→    for (const provName of PMI_SOURCE_PROVINCES) {
   702→      const idx = PROVINCES.findIndex((p) => p.name === provName);
   703→      if (idx >= 0) {
   704→        const p = buildRegionPage(svc, idx);
   705→        if (p) pages.push(p);
   706→      }
   707→    }
   708→  }
   709→
   710→  // 3. Perizinan × 10 kota besar
   711→  for (const svc of CITY_SERVICES) {
   712→    for (const city of BIG_CITIES) {
   713→      const p = buildCityPage(svc, city, "perizinan");
   714→      if (p) pages.push(p);
   715→    }
   716→  }
   717→
   718→  // 4. Pajak × 15 kota
   719→  for (const svc of TAX_CITY_SERVICES) {
   720→    for (const city of TAX_CITIES) {
   721→      const p = buildCityPage(svc, city, "pajak");
   722→      if (p) pages.push(p);
   723→    }
   724→  }
   725→
   726→  // 5. Negara PMI (17) + Negara × Sektor
   727→  for (const c of PMI_COUNTRIES) {
   728→    const cp = buildCountryPage(c.code);
   729→    if (cp) pages.push(cp);
   730→    const d = PMI_COUNTRY_MAP.get(c.code);
   731→    if (d) {
   732→      for (let i = 0; i < d.sectors.length; i++) {
   733→        const sp = buildSectorPage(c.code, i);
   734→        if (sp) pages.push(sp);
   735→      }
   736→    }
   737→  }
   738→
   739→  // 6. Virtual Office — paket × lokasi × keperluan × kota × area × provinsi × panduan (1.168)
   740→  pages.push(...VO_PAGES);
   741→
   742→  // 7. Sertifikasi — PPIU/PIHK, ISO (termasuk 9001:2026), halal jasa, pangan, lab, badan usaha (43)
   743→  pages.push(...CERT_PAGES);
   744→
   745→  // 8. Sertifikasi × 38 provinsi (1.634) + hub "Sertifikasi di {prov}" (38) = 1.672 URL
   746→  pages.push(...CERT_REGION_PAGES);
   747→  pages.push(...CERT_PROV_HUBS);
   748→
   749→  return pages;
   750→}

   764→/** Related: isi internal linking (induk, saudara wilayah, cross-category) */
   765→function buildRelated(): void {
   766→  const byCategory = new Map<CatalogCategory, ServicePage[]>();
   767→  const byRegion = new Map<string, ServicePage[]>();
   768→  for (const p of ALL_SERVICE_PAGES) {
   769→    const cat = byCategory.get(p.category) ?? [];
   770→    cat.push(p);
   771→    byCategory.set(p.category, cat);
   772→    if (p.region) {
   773→      const r = byRegion.get(p.region) ?? [];
   774→      r.push(p);
   775→      byRegion.set(p.region, r);
   776→    }
   777→  }
   778→
   779→  // Precompute indeks (O(n)) — ALL_SERVICE_PAGES kini 4.500+ halaman
   780→  const byParent = new Map<string, ServicePage[]>();
   781→  const catBase = new Map<CatalogCategory, ServicePage[]>();
   782→  for (const p of ALL_SERVICE_PAGES) {
   783→    if (p.parent) {
   784→      const arr = byParent.get(p.parent) ?? [];
   785→      arr.push(p);
   786→      byParent.set(p.parent, arr);
   787→    }
   788→    if (p.kind === "base") {
   789→      const arr = catBase.get(p.category) ?? [];
   790→      arr.push(p);
   791→      catBase.set(p.category, arr);
   792→    }
   793→  }
   794→
   795→  for (const p of ALL_SERVICE_PAGES) {
   796→    const rel: string[] = [];
   797→    // 1. Halaman induk (untuk kombinasi) / kombinasi populer (untuk induk)
   798→    if (p.parent) rel.push(p.parent);
   799→    const siblings = (byParent.get(p.parent ?? "") ?? []).filter(
   800→      (x) => x.kind === p.kind && x.slug !== p.slug
   801→    );
   802→    for (const s of siblings.slice(0, 2)) rel.push(s.slug);
   803→    // 2. Cross-category dari layanan dasar
   804→    const others = (catBase.get(p.category) ?? []).filter(
   805→      (x) => x.slug !== p.slug && x.slug !== p.parent
   806→    );
   807→    for (const o of others.slice(0, p.kind === "base" ? 3 : 2)) rel.push(o.slug);
   808→    // 4. Halaman induk sertifikasi: tautkan 3 kombinasi provinsi populer (kedalaman crawl)
   809→    if (p.kind === "base" && p.category === "sertifikasi") {
   810→      const kids = byParent.get(p.slug) ?? [];
   811→      const popular = kids.filter((k) =>
   812→        ["DKI Jakarta", "Jawa Barat", "Jawa Timur"].includes(k.region ?? "")
   813→      );
   814→      for (const k of popular.slice(0, 3)) rel.push(k.slug);
   815→    }
   816→    // 5. Sama wilayah (untuk halaman region/city)
   817→    if (p.region) {
   818→      const sameRegion = (byRegion.get(p.region) ?? []).filter((x) => x.slug !== p.slug);
   819→      for (const s of sameRegion.slice(0, 2)) rel.push(s.slug);
   820→    }
   821→    p.related = [...new Set(rel)].slice(0, 6);
   822→  }
   823→}
   824→buildRelated();

   927→/** Daftar slug hub (kategori + wilayah + kota) untuk routing & sitemap */
   928→export function getHubSlugs(): string[] {
   929→  const slugs: string[] = [];
   930→  for (const cat of Object.keys(CATEGORY_META) as CatalogCategory[]) {
   931→    if (getCategoryHub(cat)) slugs.push(`kategori/${CATEGORY_META[cat].slug}`);
   932→  }
   933→  for (let i = 0; i < PROVINCES.length; i++) {
   934→    const hub = getRegionHub(i);
   935→    if (hub) slugs.push(`wilayah/${hub.slug.split("/")[1]}`);
   936→  }
   937→  for (const cityHub of getCityHubs()) {
   938→    slugs.push(cityHub.slug);
   939→  }
   940→  return slugs;
   941→}

   943→/** Halaman kota hub (wilayah/{prov}/{kota}) — dibangun lazy setelah ALL_SERVICE_PAGES siap */
   944→export function getCityHubs(): ServicePage[] {
   945→  const pages: ServicePage[] = [];
   946→  for (let i = 0; i < PROVINCES.length; i++) {
   947→    for (const city of PROVINCES[i].majors) {
   948→      const p = buildCityHubPage(i, city);
   949→      if (p) pages.push(p);
   950→    }
   951→  }
   952→  return pages;
   953→}

   955→/** Lookup halaman apapun (termasuk hub kategori, wilayah, dan kota) */
   956→export function getAnyPage(slugOrPath: string): ServicePage | undefined {
   957→  if (PAGE_MAP.has(slugOrPath)) return PAGE_MAP.get(slugOrPath);
   958→  if (slugOrPath.startsWith("kategori/")) {
   959→    for (const cat of Object.keys(CATEGORY_META) as CatalogCategory[]) {
   960→      if (`kategori/${CATEGORY_META[cat].slug}` === slugOrPath) return getCategoryHub(cat) ?? undefined;
   961→    }
   962→  }
   963→  if (slugOrPath.startsWith("wilayah/")) {
   964→    const parts = slugOrPath.replace("wilayah/", "").split("/");
   965→    const provIdx = PROVINCES.findIndex((p) => slugify(p.name) === parts[0]);
   966→    if (provIdx < 0) return undefined;
   967→    if (parts.length === 1) return getRegionHub(provIdx) ?? undefined;
   968→    if (parts.length === 2) return buildCityHubPage(provIdx, PROVINCES[provIdx].majors.find((c) => slugify(c) === parts[1]) ?? "");
   969→  }
   970→  return undefined;
   971→}

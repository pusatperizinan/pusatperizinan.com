   661→   661→function buildAllPages(): ServicePage[] {
   662→   662→  const pages: ServicePage[] = [];

   670→   670→  // 2. Perizinan × 38 provinsi (layanan populer)
   671→   671→  for (const svc of REGION_FULL_SERVICES) {
   672→   672→    for (let i = 0; i < PROVINCES.length; i++) {
   673→   673→      const p = buildRegionPage(svc, i);
   674→   674→      if (p) pages.push(p);
   675→   675→    }
   676→   676→  }
   677→   677→  // 2b. Perizinan lain × 8 provinsi utama
   678→   678→  const liteServices = [...BASE_SERVICES.values()]
   679→   679→    .filter((b) => b.category === "perizinan" && !REGION_FULL_SERVICES.includes(b.id))
   680→   680→    .map((b) => b.id);
   681→   681→  for (const svc of liteServices) {
   682→   682→    for (const provName of REGION_LITE_PROVINCES) {
   683→   683→      const idx = PROVINCES.findIndex((p) => p.name === provName);
   684→   684→      if (idx >= 0) {
   685→   685→        const p = buildRegionPage(svc, idx);
   686→   686→        if (p) pages.push(p);
   687→   687→      }
   688→   688→    }
   689→   689→  }
   690→   690→
   691→   691→  // 2c. Pajak × 38 provinsi (layanan pajak paling dicari)
   692→   692→  for (const svc of REGION_TAX_SERVICES) {
   693→   693→    for (let i = 0; i < PROVINCES.length; i++) {
   694→   694→      const p = buildRegionPage(svc, i);
   695→   695→      if (p) pages.push(p);
   696→   696→    }
   697→   697→  }
   698→   698→
   699→   699→  // 2d. PMI B2C × 8 provinsi utama pengirim migran
   700→   700→  for (const svc of REGION_PMI_SERVICES) {
   701→   701→    for (const provName of PMI_SOURCE_PROVINCES) {
   702→   702→      const idx = PROVINCES.findIndex((p) => p.name === provName);
   703→   703→      if (idx >= 0) {
   704→   704→        const p = buildRegionPage(svc, idx);
   705→   705→        if (p) pages.push(p);
   706→   706→      }
   707→   707→    }
   708→   708→  }
   709→   709→
   710→   710→  // 3. Perizinan × 10 kota besar
   711→   711→  for (const svc of CITY_SERVICES) {
   712→   712→    for (const city of BIG_CITIES) {
   713→   713→      const p = buildCityPage(svc, city, "perizinan");
   714→   714→      if (p) pages.push(p);
   715→   715→    }
   716→   716→  }
   717→   717→
   718→   718→  // 4. Pajak × 15 kota
   719→   719→  for (const svc of TAX_CITY_SERVICES) {
   720→   720→    for (const city of TAX_CITIES) {
   721→   721→      const p = buildCityPage(svc, city, "pajak");

   723→   723→    }
   724→   724→  }
   725→   725→
   726→   726→  // 5. Negara PMI (17) + Negara × Sektor
   727→   727→  for (const c of PMI_COUNTRIES) {
   728→   728→    const cp = buildCountryPage(c.code);
   729→   729→    if (cp) pages.push(cp);
   730→   730→    const d = PMI_COUNTRY_MAP.get(c.code);
   731→   731→    if (d) {
   732→   732→      for (let i = 0; i < d.sectors.length; i++) {
   733→   733→        const sp = buildSectorPage(c.code, i);
   734→   734→        if (sp) pages.push(sp);
   735→   735→      }
   736→   736→    }
   737→   737→  }
   738→   738→
   739→   739→  // 6. Virtual Office — paket × lokasi × keperluan × kota × area × provinsi × panduan (1.168)
   740→   740→  pages.push(...VO_PAGES);
   741→   741→
   742→   742→  // 7. Sertifikasi — PPIU/PIHK, ISO (termasuk 9001:2026), halal jasa, pangan, lab, badan usaha (43)
   743→   743→  pages.push(...CERT_PAGES);
   744→   744→
   745→   745→  // 8. Sertifikasi × 38 provinsi (1.634) + hub "Sertifikasi di {prov}" (38) = 1.672 URL
   746→   746→  pages.push(...CERT_REGION_PAGES);
   747→   747→  pages.push(...CERT_PROV_HUBS);
   748→   748→
   749→   749→  return pages;
   750→   750→}

   764→   764→/** Related: isi internal linking (induk, saudara wilayah, cross-category) */
   765→   765→function buildRelated(): void {
   766→   766→  const byCategory = new Map<CatalogCategory, ServicePage[]>();
   767→   767→  const byRegion = new Map<string, ServicePage[]>();
   768→   768→  for (const p of ALL_SERVICE_PAGES) {
   769→   769→    const cat = byCategory.get(p.category) ?? [];
   770→   770→    cat.push(p);
   771→   771→    byCategory.set(p.category, cat);
   772→   772→    if (p.region) {
   773→   773→      const r = byRegion.get(p.region) ?? [];
   774→   774→      r.push(p);
   775→   775→      byRegion.set(p.region, r);
   776→   776→    }
   777→   777→  }
   778→   778→
   779→   779→  // Precompute indeks (O(n)) — ALL_SERVICE_PAGES kini 4.500+ halaman
   780→   780→  const byParent = new Map<string, ServicePage[]>();
   781→   781→  const catBase = new Map<CatalogCategory, ServicePage[]>();
   782→   782→  for (const p of ALL_SERVICE_PAGES) {
   783→   783→    if (p.parent) {
   784→   784→      const arr = byParent.get(p.parent) ?? [];
   785→   785→      arr.push(p);
   786→   786→      byParent.set(p.parent, arr);
   787→   787→    }
   788→   788→    if (p.kind === "base") {
   789→   789→      const arr = catBase.get(p.category) ?? [];
   790→   790→      arr.push(p);
   791→   791→      catBase.set(p.category, arr);
   792→   792→    }
   793→   793→  }
   794→   794→
   795→   795→  for (const p of ALL_SERVICE_PAGES) {
   796→   796→    const rel: string[] = [];
   797→   797→    // 1. Halaman induk (untuk kombinasi) / kombinasi populer (untuk induk)
   798→   798→    if (p.parent) rel.push(p.parent);
   799→   799→    const siblings = (byParent.get(p.parent ?? "") ?? []).filter(
   800→   800→      (x) => x.kind === p.kind && x.slug !== p.slug
   801→   801→    );
   802→   802→    for (const s of siblings.slice(0, 2)) rel.push(s.slug);
   803→   803→    // 2. Cross-category dari layanan dasar
   804→   804→    const others = (catBase.get(p.category) ?? []).filter(
   805→   805→      (x) => x.slug !== p.slug && x.slug !== p.parent
   806→   806→    );
   807→   807→    for (const o of others.slice(0, p.kind === "base" ? 3 : 2)) rel.push(o.slug);
   808→   808→    // 4. Halaman induk sertifikasi: tautkan 3 kombinasi provinsi populer (kedalaman crawl)
   809→   809→    if (p.kind === "base" && p.category === "sertifikasi") {
   810→   810→      const kids = byParent.get(p.slug) ?? [];
   811→   811→      const popular = kids.filter((k) =>
   812→   812→        ["DKI Jakarta", "Jawa Barat", "Jawa Timur"].includes(k.region ?? "")
   813→   813→      );
   814→   814→      for (const k of popular.slice(0, 3)) rel.push(k.slug);
   815→   815→    }
   816→   816→    // 5. Sama wilayah (untuk halaman region/city)
   817→   817→    if (p.region) {
   818→   818→      const sameRegion = (byRegion.get(p.region) ?? []).filter((x) => x.slug !== p.slug);
   819→   819→      for (const s of sameRegion.slice(0, 2)) rel.push(s.slug);
   820→   820→    }
   821→   821→    p.related = [...new Set(rel)].slice(0, 6);
   822→   822→  }
   823→   823→}
   824→   824→buildRelated();

   928→   928→export function getHubSlugs(): string[] {
   929→   929→  const slugs: string[] = [];
   930→   930→  for (const cat of Object.keys(CATEGORY_META) as CatalogCategory[]) {
   931→   931→    if (getCategoryHub(cat)) slugs.push(`kategori/${CATEGORY_META[cat].slug}`);
   932→   932→  }
   933→   933→  for (let i = 0; i < PROVINCES.length; i++) {
   934→   934→    const hub = getRegionHub(i);
   935→   935→    if (hub) slugs.push(`wilayah/${hub.slug.split("/")[1]}`);
   936→   936→  }
   937→   937→  for (const cityHub of getCityHubs()) {
   938→   938→    slugs.push(cityHub.slug);
   939→   939→  }
   940→   940→  return slugs;
   941→   941→}

   955→   955→/** Lookup halaman apapun (termasuk hub kategori, wilayah, dan kota) */
   956→   956→export function getAnyPage(slugOrPath: string): ServicePage | undefined {
   957→   957→  if (PAGE_MAP.has(slugOrPath)) return PAGE_MAP.get(slugOrPath);
   958→   958→  if (slugOrPath.startsWith("kategori/")) {
   959→   959→    for (const cat of Object.keys(CATEGORY_META) as CatalogCategory[]) {
   960→   960→      if (`kategori/${CATEGORY_META[cat].slug}` === slugOrPath) return getCategoryHub(cat) ?? undefined;
   961→   961→    }
   962→   962→  }
   963→   963→  if (slugOrPath.startsWith("wilayah/")) {
   964→   964→    const parts = slugOrPath.replace("wilayah/", "").split("/");
   965→   965→    const provIdx = PROVINCES.findIndex((p) => slugify(p.name) === parts[0]);
   966→   966→    if (provIdx < 0) return undefined;
   967→   967→    if (parts.length === 1) return getRegionHub(provIdx) ?? undefined;
   968→   968→    if (parts.length === 2) return buildCityHubPage(provIdx, PROVINCES[provIdx].majors.find((c) => slugify(c) === parts[1]) ?? "");
   969→   969→  }
   970→   970→  return undefined;
   971→   971→}

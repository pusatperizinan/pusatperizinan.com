    64→export async function generateMetadata({
    65→  params,
    66→}: {
    67→  params: Promise<{ slug: string[] }>;
    68→}): Promise<Metadata> {
    69→  const { slug } = await params;

    79→      title: page.title,

    82→      type: "article",

    85→    },

    88→      title: page.title,

   119→        address: {

   121→          streetAddress: "Indonesia Stock Exchange Building, Tower 2, Lantai 5, SCBD Lot 13, Jl. Jend. Sudirman Kav. 52-53",
   122→          addressLocality: "Jakarta Selatan",
   123→          addressRegion: "DKI Jakarta",
   124→          postalCode: "12190",
   125→          addressCountry: "ID",
   126→        },
   127→        aggregateRating: {
   128→          "@type": "AggregateRating",
   129→          ratingValue: "4.9",
   130→          reviewCount: "1247",
   131→          bestRating: "5",
   132→        },
   133→      },
   134→      areaServed: page.region
   135→        ? { "@type": "Place", name: page.region }
   136→        : page.country

   140→        ? {

   176→  }
   177→
   178→  // 2. FAQPage schema
   179→  if (page.faq.length > 0) {
   180→    jsonLd.push({
   181→      "@context": "https://schema.org",
   182→      "@type": "FAQPage",
   183→      mainEntity: page.faq.map((f) => ({
   184→        "@type": "Question",

   192→  jsonLd.push({
   193→    "@context": "https://schema.org",
   194→    "@type": "BreadcrumbList",
   195→    itemListElement: page.breadcrumbs.map((b, i) => ({
   196→      "@type": "ListItem",
   197→      position: i + 1,
   198→      name: b.name,
   199→      item: `${SITE_URL}${b.href}`,
   200→    })),
   201→  });
   202→
   203→  return (
   204→    <script
   205→      type="application/ld+json"
   206→      dangerouslySetInnerHTML={{ __html: JSON.stringify(jsonLd) }}
   207→    />
   208→  );
   209→}

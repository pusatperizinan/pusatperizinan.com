    76→export function slugify(text: string): string {
    77→  return text
    78→    .toLowerCase()
    79→    .trim()
    80→    .replace(/\s+/g, "-")
    81→    .replace(/[^a-z0-9-]/g, "");
    82→}
    83→
    84→/** Potong metaDesc agar tidak melebihi batas mesin pencari dengan rapi */
    85→function capMeta(text: string, max = 300): string {
    86→  if (text.length <= max) return text;
    87→  const cut = text.slice(0, max);
    88→  const lastStop = Math.max(cut.lastIndexOf(". "), cut.lastIndexOf("— "), cut.lastIndexOf(", "));
    89→  return (lastStop > max * 0.6 ? cut.slice(0, lastStop) : cut.replace(/[,;\s]+$/, "")) + ".";
    90→}

    92→export function parsePrice(price: string): number {
    93→  const clean = price.replace(/\./g, "").toLowerCase();
    94→  const match = clean.match(/(\d+)(?:,(\d+))?\s*(jt|rb|juta|ribu)?/);
    95→  if (!match) return 0;
    96→  const num = parseInt(match[1] || "0", 10);
    97→  const frac = match[2] ? parseInt(match[2], 10) : 0;
    98→  let val = num + (frac ? frac / Math.pow(10, match[2]?.length || 1) : 0);
    99→  const unit = match[3];
   100→  if (unit === "jt" || unit === "juta") val *= 1_000_000;
   101→  else if (unit === "rb" || unit === "ribu") val *= 1_000;
   102→  return Math.round(val);
   103→}

   671→// ------------------------------------------------------------
   672→// BUILD ALL — agregasi seluruh katalog
   673→// ------------------------------------------------------------
   674→
   675→function buildAllPages(): ServicePage[] {
   676→  const pages: ServicePage[] = [];
   677→
   678→  // 1. Halaman induk (61 layanan)
   679→  for (const id of BASE_SERVICES.keys()) {
   680→    const p = buildBasePage(id);
   681→    if (p) pages.push(p);
   682→  }
   683→
   684→  // 2. Perizinan × 38 provinsi (layanan populer)
   685→  for (const svc of REGION_FULL_SERVICES) {
   686→    for (let i = 0; i < PROVINCES.length; i++) {
   687→      const p = buildRegionPage(svc, i);
   688→      if (p) pages.push(p);
   689→    }
   690→  }
   691→  // 2b. Perizinan lain × 8 provinsi utama
   692→  const liteServices = [...BASE_SERVICES.values()]
   693→    .filter((b) => b.category === "perizinan" && !REGION_FULL_SERVICES.includes(b.id))
   694→    .map((b) => b.id);
   695→  for (const svc of liteServices) {
   696→    for (const provName of REGION_LITE_PROVINCES) {
   697→      const idx = PROVINCES.findIndex((p) => p.name === provName);
   698→      if (idx >= 0) {
   699→        const p = buildRegionPage(svc, idx);
   700→        if (p) pages.push(p);
   701→      }
   702→    }
   703→  }
   704→
   705→  // 2c. Pajak × 38 provinsi (layanan pajak paling dicari)
   706→  for (const svc of REGION_TAX_SERVICES) {
   707→    for (let i = 0; i < PROVINCES.length; i++) {
   708→      const p = buildRegionPage(svc, i);
   709→      if (p) pages.push(p);
   710→    }
   711→  }
   712→
   713→  // 2d. PMI B2C × 8 provinsi utama pengirim migran
   714→  for (const svc of REGION_PMI_SERVICES) {
   715→    for (const provName of PMI_SOURCE_PROVINCES) {
   716→      const idx = PROVINCES.findIndex((p) => p.name === provName);
   717→      if (idx >= 0) {
   718→        const p = buildRegionPage(svc, idx);
   719→        if (p) pages.push(p);
   720→      }
   721→    }
   722→  }
   723→
   724→  // 3. Perizinan × 10 kota besar
   725→  for (const svc of CITY_SERVICES) {
   726→    for (const city of BIG_CITIES) {
   727→      const p = buildCityPage(svc, city, "perizinan");
   728→      if (p) pages.push(p);
   729→    }
   730→  }
   731→
   732→  // 4. Pajak × 15 kota
   733→  for (const svc of TAX_CITY_SERVICES) {
   734→    for (const city of TAX_CITIES) {
   735→      const p = buildCityPage(svc, city, "pajak");
   736→      if (p) pages.push(p);
   737→    }
   738→  }
   739→
   740→  // 5. Negara PMI (17) + Negara × Sektor
   741→  for (const c of PMI_COUNTRIES) {
   742→    const cp = buildCountryPage(c.code);
   743→    if (cp) pages.push(cp);
   744→    const d = PMI_COUNTRY_MAP.get(c.code);
   745→    if (d) {
   746→      for (let i = 0; i < d.sectors.length; i++) {
   747→        const sp = buildSectorPage(c.code, i);
   748→        if (sp) pages.push(sp);
   749→      }
   750→    }
   751→  }
   752→
   753→  return pages;
   754→}
   755→

   768→/** Related: isi internal linking (induk, saudara wilayah, cross-category) */
   769→function buildRelated(): void {
   770→  const byCategory = new Map<CatalogCategory, ServicePage[]>();
   771→  const byRegion = new Map<string, ServicePage[]>();
   772→  for (const p of ALL_SERVICE_PAGES) {
   773→    const cat = byCategory.get(p.category) ?? [];
   774→    cat.push(p);
   775→    byCategory.set(p.category, cat);
   776→    if (p.region) {
   777→      const r = byRegion.get(p.region) ?? [];
   778→      r.push(p);
   779→      byRegion.set(p.region, r);
   780→    }
   781→  }
   782→
   783→  for (const p of ALL_SERVICE_PAGES) {
   784→    const rel: string[] = [];
   785→    // 1. Halaman induk (untuk kombinasi) / kombinasi populer (untuk induk)
   786→    if (p.parent) rel.push(p.parent);
   787→    const siblings = ALL_SERVICE_PAGES.filter(
   788→      (x) => x.parent === p.parent && x.kind === p.kind && x.slug !== p.slug
   789→    );
   790→    for (const s of siblings.slice(0, 2)) rel.push(s.slug);
   791→    // 2. Cross-category dari layanan dasar
   792→    const others = ALL_SERVICE_PAGES.filter(
   793→      (x) => x.kind === "base" && x.category === p.category && x.slug !== p.slug && x.slug !== p.parent
   794→    );
   795→    for (const o of others.slice(0, p.kind === "base" ? 3 : 2)) rel.push(o.slug);
   796→    // 3. Sama wilayah (untuk halaman region/city)
   797→    if (p.region) {
   798→      const sameRegion = (byRegion.get(p.region) ?? []).filter((x) => x.slug !== p.slug);
   799→      for (const s of sameRegion.slice(0, 2)) rel.push(s.slug);
   800→    }
   801→    p.related = [...new Set(rel)].slice(0, 6);
   802→  }
   803→}
   804→buildRelated();

   907→/** Daftar slug hub (kategori + wilayah + kota) untuk routing & sitemap */
   908→export function getHubSlugs(): string[] {
   909→  const slugs: string[] = [];
   910→  for (const cat of Object.keys(CATEGORY_META) as CatalogCategory[]) {
   911→    if (getCategoryHub(cat)) slugs.push(`kategori/${CATEGORY_META[cat].slug}`);
   912→  }
   913→  for (let i = 0; i < PROVINCES.length; i++) {
   914→    const hub = getRegionHub(i);
   915→    if (hub) slugs.push(`wilayah/${hub.slug.split("/")[1]}`);
   916→  }
   917→  for (const cityHub of getCityHubs()) {
   918→    slugs.push(cityHub.slug);
   919→  }
   920→  return slugs;
   921→}

   923→/** Halaman kota hub (wilayah/{prov}/{kota}) — dibangun lazy setelah ALL_SERVICE_PAGES siap */
   924→export function getCityHubs(): ServicePage[] {
   925→  const pages: ServicePage[] = [];
   926→  for (let i = 0; i < PROVINCES.length; i++) {
   927→    for (const city of PROVINCES[i].majors) {
   928→      const p = buildCityHubPage(i, city);
   929→      if (p) pages.push(p);
   930→    }
   931→  }
   932→  return pages;
   933→}

   935→/** Lookup halaman apapun (termasuk hub kategori, wilayah, dan kota) */
   936→export function getAnyPage(slugOrPath: string): ServicePage | undefined {
   937→  if (PAGE_MAP.has(slugOrPath)) return PAGE_MAP.get(slugOrPath);
   938→  if (slugOrPath.startsWith("kategori/")) {
   939→    for (const cat of Object.keys(CATEGORY_META) as CatalogCategory[]) {
   940→      if (`kategori/${CATEGORY_META[cat].slug}` === slugOrPath) return getCategoryHub(cat) ?? undefined;
   941→    }
   942→  }
   943→  if (slugOrPath.startsWith("wilayah/")) {
   944→    const parts = slugOrPath.replace("wilayah/", "").split("/");
   945→    const provIdx = PROVINCES.findIndex((p) => slugify(p.name) === parts[0]);
   946→    if (provIdx < 0) return undefined;
   947→    if (parts.length === 1) return getRegionHub(provIdx) ?? undefined;
   948→    if (parts.length === 2) return buildCityHubPage(provIdx, PROVINCES[provIdx].majors.find((c) => slugify(c) === parts[1]) ?? "");
   949→  }
   950→  return undefined;
   951→}

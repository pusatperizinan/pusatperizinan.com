    59→export async function generateMetadata({
    60→  params,
    61→}: {
    62→  params: Promise<{ slug: string[] }>;
    63→}): Promise<Metadata> {
    64→  const { slug } = await params;
    65→  const page = getAnyPage(slug.join("/"));

    77→      type: "article",

    83→      title: page.title,

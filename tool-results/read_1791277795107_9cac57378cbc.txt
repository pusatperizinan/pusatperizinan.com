    81→/** Potong metaDesc agar tidak melebihi batas mesin pencari dengan rapi */
    82→function capMeta(text: string, max = 300): string {
    83→  if (text.length <= max) return text;
    84→  const cut = text.slice(0, max);
    85→  const lastStop = Math.max(cut.lastIndexOf(". "), cut.lastIndexOf("— "), cut.lastIndexOf(", "));
    86→  return (lastStop > max * 0.6 ? cut.slice(0, lastStop) : cut.replace(/[,;\s]+$/, "")) + ".";
    87→}

   655→// ------------------------------------------------------------
   656→// BUILD ALL — agregasi seluruh katalog
   657→// ------------------------------------------------------------
   658→
   659→function buildAllPages(): ServicePage[] {
   660→  const pages: ServicePage[] = [];
   661→
   662→  // 1. Halaman induk (61 layanan)
   663→  for (const id of BASE_SERVICES.keys()) {
   664→    const p = buildBasePage(id);
   665→    if (p) pages.push(p);
   666→  }
   667→
   668→  // 2. Perizinan × 38 provinsi (layanan populer)
   669→  for (const svc of REGION_FULL_SERVICES) {
   670→    for (let i = 0; i < PROVINCES.length; i++) {
   671→      const p = buildRegionPage(svc, i);
   672→      if (p) pages.push(p);
   673→    }
   674→  }
   675→  // 2b. Perizinan lain × 8 provinsi utama
   676→  const liteServices = [...BASE_SERVICES.values()]
   677→    .filter((b) => b.category === "perizinan" && !REGION_FULL_SERVICES.includes(b.id))
   678→    .map((b) => b.id);
   679→  for (const svc of liteServices) {
   680→    for (const provName of REGION_LITE_PROVINCES) {
   681→      const idx = PROVINCES.findIndex((p) => p.name === provName);
   682→      if (idx >= 0) {
   683→        const p = buildRegionPage(svc, idx);
   684→        if (p) pages.push(p);
   685→      }
   686→    }
   687→  }
   688→
   689→  // 2c. Pajak × 38 provinsi (layanan pajak paling dicari)
   690→  for (const svc of REGION_TAX_SERVICES) {
   691→    for (let i = 0; i < PROVINCES.length; i++) {
   692→      const p = buildRegionPage(svc, i);
   693→      if (p) pages.push(p);
   694→    }
   695→  }
   696→
   697→  // 2d. PMI B2C × 8 provinsi utama pengirim migran
   698→  for (const svc of REGION_PMI_SERVICES) {
   699→    for (const provName of PMI_SOURCE_PROVINCES) {
   700→      const idx = PROVINCES.findIndex((p) => p.name === provName);
   701→      if (idx >= 0) {
   702→        const p = buildRegionPage(svc, idx);
   703→        if (p) pages.push(p);
   704→      }
   705→    }
   706→  }
   707→
   708→  // 3. Perizinan × 10 kota besar
   709→  for (const svc of CITY_SERVICES) {
   710→    for (const city of BIG_CITIES) {
   711→      const p = buildCityPage(svc, city, "perizinan");
   712→      if (p) pages.push(p);
   713→    }
   714→  }
   715→
   716→  // 4. Pajak × 15 kota
   717→  for (const svc of TAX_CITY_SERVICES) {
   718→    for (const city of TAX_CITIES) {
   719→      const p = buildCityPage(svc, city, "pajak");
   720→      if (p) pages.push(p);
   721→    }
   722→  }
   723→
   724→  // 5. Negara PMI (17) + Negara × Sektor
   725→  for (const c of PMI_COUNTRIES) {
   726→    const cp = buildCountryPage(c.code);
   727→    if (cp) pages.push(cp);
   728→    const d = PMI_COUNTRY_MAP.get(c.code);
   729→    if (d) {
   730→      for (let i = 0; i < d.sectors.length; i++) {
   731→        const sp = buildSectorPage(c.code, i);
   732→        if (sp) pages.push(sp);
   733→      }
   734→    }
   735→  }
   736→
   737→  // 6. Virtual Office — paket × lokasi × keperluan × kota × area × provinsi × panduan (1.168)
   738→  pages.push(...VO_PAGES);
   739→
   740→  // 7. Sertifikasi — PPIU/PIHK, ISO (termasuk 9001:2026), halal jasa, pangan, lab, badan usaha (43)
   741→  pages.push(...CERT_PAGES);
   742→
   743→  return pages;
   744→}

   752→}

   758→/** Related: isi internal linking (induk, saudara wilayah, cross-category) */
   759→function buildRelated(): void {
   760→  const byCategory = new Map<CatalogCategory, ServicePage[]>();
   761→  const byRegion = new Map<string, ServicePage[]>();
   762→  for (const p of ALL_SERVICE_PAGES) {
   763→    const cat = byCategory.get(p.category) ?? [];
   764→    cat.push(p);
   765→    byCategory.set(p.category, cat);
   766→    if (p.region) {
   767→      const r = byRegion.get(p.region) ?? [];
   768→      r.push(p);
   769→      byRegion.set(p.region, r);
   770→    }
   771→  }
   772→
   773→  for (const p of ALL_SERVICE_PAGES) {
   774→    const rel: string[] = [];
   775→    // 1. Halaman induk (untuk kombinasi) / kombinasi populer (untuk induk)
   776→    if (p.parent) rel.push(p.parent);
   777→    const siblings = ALL_SERVICE_PAGES.filter(
   778→      (x) => x.parent === p.parent && x.kind === p.kind && x.slug !== p.slug
   779→    );
   780→    for (const s of siblings.slice(0, 2)) rel.push(s.slug);
   781→    // 2. Cross-category dari layanan dasar
   782→    const others = ALL_SERVICE_PAGES.filter(
   783→      (x) => x.kind === "base" && x.category === p.category && x.slug !== p.slug && x.slug !== p.parent
   784→    );
   785→    for (const o of others.slice(0, p.kind === "base" ? 3 : 2)) rel.push(o.slug);
   786→    // 3. Sama wilayah (untuk halaman region/city)
   787→    if (p.region) {
   788→      const sameRegion = (byRegion.get(p.region) ?? []).filter((x) => x.slug !== p.slug);
   789→      for (const s of sameRegion.slice(0, 2)) rel.push(s.slug);
   790→    }
   791→    p.related = [...new Set(rel)].slice(0, 6);
   792→  }
   793→}
   794→buildRelated();

   897→/** Daftar slug hub (kategori + wilayah + kota) untuk routing & sitemap */
   898→export function getHubSlugs(): string[] {
   899→  const slugs: string[] = [];
   900→  for (const cat of Object.keys(CATEGORY_META) as CatalogCategory[]) {
   901→    if (getCategoryHub(cat)) slugs.push(`kategori/${CATEGORY_META[cat].slug}`);
   902→  }
   903→  for (let i = 0; i < PROVINCES.length; i++) {
   904→    const hub = getRegionHub(i);
   905→    if (hub) slugs.push(`wilayah/${hub.slug.split("/")[1]}`);
   906→  }
   907→  for (const cityHub of getCityHubs()) {
   908→    slugs.push(cityHub.slug);
   909→  }
   910→  return slugs;
   911→}

   913→/** Halaman kota hub (wilayah/{prov}/{kota}) — dibangun lazy setelah ALL_SERVICE_PAGES siap */
   914→export function getCityHubs(): ServicePage[] {
   915→  const pages: ServicePage[] = [];
   916→  for (let i = 0; i < PROVINCES.length; i++) {
   917→    for (const city of PROVINCES[i].majors) {
   918→      const p = buildCityHubPage(i, city);
   919→      if (p) pages.push(p);
   920→    }
   921→  }
   922→  return pages;
   923→}

   925→/** Lookup halaman apapun (termasuk hub kategori, wilayah, dan kota) */
   926→export function getAnyPage(slugOrPath: string): ServicePage | undefined {
   927→  if (PAGE_MAP.has(slugOrPath)) return PAGE_MAP.get(slugOrPath);
   928→  if (slugOrPath.startsWith("kategori/")) {
   929→    for (const cat of Object.keys(CATEGORY_META) as CatalogCategory[]) {
   930→      if (`kategori/${CATEGORY_META[cat].slug}` === slugOrPath) return getCategoryHub(cat) ?? undefined;
   931→    }
   932→  }
   933→  if (slugOrPath.startsWith("wilayah/")) {
   934→    const parts = slugOrPath.replace("wilayah/", "").split("/");
   935→    const provIdx = PROVINCES.findIndex((p) => slugify(p.name) === parts[0]);
   936→    if (provIdx < 0) return undefined;
   937→    if (parts.length === 1) return getRegionHub(provIdx) ?? undefined;
   938→    if (parts.length === 2) return buildCityHubPage(provIdx, PROVINCES[provIdx].majors.find((c) => slugify(c) === parts[1]) ?? "");
   939→  }
   940→  return undefined;
   941→}

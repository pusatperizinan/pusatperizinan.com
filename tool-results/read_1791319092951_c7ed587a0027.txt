    53→/** Hash deterministik (FNV-1a 32-bit) untuk variasi konten per-slug.
    54→ *  Tujuan: halaman sejenis tidak identik satu sama lain (anti-doorway,
    55→ *  lolos ambang similarity audit) tanpa perlu AI/database. */
    56→function hashStr(s: string): number {
    57→  let h = 2166136261;
    58→  for (let i = 0; i < s.length; i++) {
    59→    h ^= s.charCodeAt(i);
    60→    h = Math.imul(h, 16777619);
    61→  }
    62→  return h >>> 0;
    63→}

    65→/** Potong metaDesc agar tidak melebihi batas mesin pencari dengan rapi */
    66→function capMeta(text: string, max = 300): string {
    67→  if (text.length <= max) return text;
    68→  const cut = text.slice(0, max);
    69→  const lastStop = Math.max(cut.lastIndexOf(". "), cut.lastIndexOf("— "), cut.lastIndexOf(", "));
    70→  return (lastStop > max * 0.6 ? cut.slice(0, lastStop) : cut.replace(/[,;\s]+$/, "")) + ".";
    71→}

  1069→/** Related: isi internal linking (induk, saudara wilayah, cross-category) */
  1070→function buildRelated(): void {
  1071→  const byCategory = new Map<CatalogCategory, ServicePage[]>();
  1072→  const byRegion = new Map<string, ServicePage[]>();
  1073→  for (const p of ALL_SERVICE_PAGES) {
  1074→    const cat = byCategory.get(p.category) ?? [];
  1075→    cat.push(p);
  1076→    byCategory.set(p.category, cat);
  1077→    if (p.region) {
  1078→      const r = byRegion.get(p.region) ?? [];
  1079→      r.push(p);
  1080→      byRegion.set(p.region, r);
  1081→    }
  1082→  }
  1083→
  1084→  // Precompute indeks (O(n)) — ALL_SERVICE_PAGES kini 4.500+ halaman
  1085→  const byParent = new Map<string, ServicePage[]>();
  1086→  const catBase = new Map<CatalogCategory, ServicePage[]>();
  1087→  for (const p of ALL_SERVICE_PAGES) {
  1088→    if (p.parent) {
  1089→      const arr = byParent.get(p.parent) ?? [];
  1090→      arr.push(p);
  1091→      byParent.set(p.parent, arr);
  1092→    }
  1093→    if (p.kind === "base") {
  1094→      const arr = catBase.get(p.category) ?? [];
  1095→      arr.push(p);
  1096→      catBase.set(p.category, arr);
  1097→    }
  1098→  }
  1099→
  1100→  for (const p of ALL_SERVICE_PAGES) {
  1101→    const rel: string[] = [];
  1102→    // 1. Halaman induk (untuk kombinasi) / kombinasi populer (untuk induk)
  1103→    if (p.parent) rel.push(p.parent);
  1104→    const siblings = (byParent.get(p.parent ?? "") ?? []).filter(
  1105→      (x) => x.kind === p.kind && x.slug !== p.slug
  1106→    );
  1107→    for (const s of siblings.slice(0, 2)) rel.push(s.slug);
  1108→    // 2. Cross-category dari layanan dasar
  1109→    const others = (catBase.get(p.category) ?? []).filter(
  1110→      (x) => x.slug !== p.slug && x.slug !== p.parent
  1111→    );
  1112→    for (const o of others.slice(0, p.kind === "base" ? 3 : 2)) rel.push(o.slug);
  1113→    // 4. Halaman induk sertifikasi: tautkan 3 kombinasi provinsi populer (kedalaman crawl)
  1114→    if (p.kind === "base" && p.category === "sertifikasi") {
  1115→      const kids = byParent.get(p.slug) ?? [];
  1116→      const popular = kids.filter((k) =>
  1117→        ["DKI Jakarta", "Jawa Barat", "Jawa Timur"].includes(k.region ?? "")
  1118→      );
  1119→      for (const k of popular.slice(0, 3)) rel.push(k.slug);
  1120→    }
  1121→    // 5. Sama wilayah (untuk halaman region/city)
  1122→    if (p.region) {
  1123→      const sameRegion = (byRegion.get(p.region) ?? []).filter((x) => x.slug !== p.slug);
  1124→      for (const s of sameRegion.slice(0, 2)) rel.push(s.slug);
  1125→    }
  1126→    p.related = [...new Set(rel)].slice(0, 6);
  1127→  }
  1128→}

  1260→/** Lookup halaman apapun (termasuk hub kategori, wilayah, dan kota) */
  1261→export function getAnyPage(slugOrPath: string): ServicePage | undefined {
  1262→  if (PAGE_MAP.has(slugOrPath)) return PAGE_MAP.get(slugOrPath);
  1263→  if (slugOrPath.startsWith("kategori/")) {
  1264→    for (const cat of Object.keys(CATEGORY_META) as CatalogCategory[]) {
  1265→      if (`kategori/${CATEGORY_META[cat].slug}` === slugOrPath) return getCategoryHub(cat) ?? undefined;
  1266→    }
  1267→  }
  1268→  if (slugOrPath.startsWith("wilayah/")) {
  1269→    const parts = slugOrPath.replace("wilayah/", "").split("/");
  1270→    const provIdx = PROVINCES.findIndex((p) => slugify(p.name) === parts[0]);
  1271→    if (provIdx < 0) return undefined;
  1272→    if (parts.length === 1) return getRegionHub(provIdx) ?? undefined;
  1273→    if (parts.length === 2) return buildCityHubPage(provIdx, PROVINCES[provIdx].majors.find((c) => slugify(c) === parts[1]) ?? "");
  1274→  }
  1275→  return undefined;
  1276→}
